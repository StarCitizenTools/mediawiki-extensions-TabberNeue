local p = {}
function p.main()
	return mw.ext.tabber.render(
		{ { label = 'Alpha', content = 'A content' } },
		{ id = 'lua-id', class = 'lua-class', ['data-x'] = 'lua-data', wrap = true }
	)
end
function p.unsafe()
	return mw.ext.tabber.render(
		{ { label = 'Alpha', content = 'A content' } },
		{ onmouseover = 'alert(1)', style = 'background:url(javascript:alert(1))', id = 'a"><script>alert(1)</script>' }
	)
end
function p.noattrs()
	return mw.ext.tabber.render( { { label = 'Alpha', content = 'A content' } } )
end
return p
!! endarticle
